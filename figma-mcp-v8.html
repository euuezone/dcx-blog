function ProgressRing({ percent, color, bgColor, size=42 }) {
  const r = size/2-5;
  const circ = 2*Math.PI*r;
  const offset = circ*(1-percent/100);
  return (
    <div style={{position:"relative",width:size,height:size}}>
      <svg width={size} height={size}>
        <circle cx={size/2} cy={size/2} r={r} fill={bgColor}/>
        <circle cx={size/2} cy={size/2} r={r} fill="none"
          stroke={color} strokeWidth="5"
          strokeDasharray={`${circ} ${circ}`}
          strokeDashoffset={offset}
          transform={`rotate(-90 ${size/2} ${size/2})`}/>
      </svg>
      <span style={{position:"absolute",fontSize:11}}>{percent}%</span>
    </div>
  );
}

🏁 결론

테스트 1 — Figma 수정 자동화: ClaudeTalkToFigma 우세 — 텍스트 한글화 + 색상 변경을 한 번에, API 한도 없이 처리. 84개 쓰기 명령, 토큰 최소(~4,500t/화면).

추가 테스트 — Figma 비교 보드 자동 생성: upload_assets로 Figma 캔버스에 직접 이미지 삽입 가능. CTF 행 완성 후 유료 플랜에서 전체 3행 비교 보드를 한 번에 완성 가능.

테스트 2 — 코드 생성: 공식 Figma MCP 우위 — React + Tailwind CSS + 이미지 에셋 URL 28개를 한 번에 출력. GLips MCP도 구조화 텍스트로 충분한 컨텍스트를 제공하지만 이미지 포함 여부에서 차이남.

결론: 목적에 맞게 선택. Figma 텍스트·색상 수정 자동화 → CTF · 코드 생성 + 에셋 → 공식 MCP · 가벼운 구조 파악 → GLips
